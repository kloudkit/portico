```sh
# TL-DR; 🚪 One port in front of every service
curl -fsSL -o ~/.local/bin/portico \
  https://github.com/kloudkit/portico/releases/latest/download/portico-amd64
chmod +x ~/.local/bin/portico

portico init && portico up
```
